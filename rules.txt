mv (arquivo) (pasta existente) -> move o arquivo pra pasta | OK

mv (pasta) (arquivo existente) -> erro | OK

cp (arquivo) (arquivo n existente) -> criar copia do arquivo com nome do arquivo2 OK

cp (arquivo) (pasta) -> copia arquivo pra dentro da pasta OK

cp (pasta) (arquivo) -> erro OK

n1 -> permissoes do user
n2 -> permissoes do grupo
n3 -> permissoes de outros

r = 4 (read)
w = 2 (write)
x = 1 (execute)

chmod 040 arquivo -> permissao de leitura no arquivo para (grupo)
chmod 004 arquivo -> permissao de leitura no arquivo para (outros)
